5. 最核心的 Unity Shader 思路
像素锁定采样函数
float2 PixelSnapUV(float2 uv, float2 resolution)
{
    float2 pixel = floor(uv * resolution);
    return (pixel + 0.5) / resolution;
}

所有全屏后处理的 screen sample 都优先走这个函数。

色差：整数像素偏移
float4 PixelChromatic(float2 uv)
{
    float2 res = float2(320, 180);
    float2 p = floor(uv * res);

    float2 rUV = (p + float2( 1, 0) + 0.5) / res;
    float2 gUV = (p + float2( 0, 0) + 0.5) / res;
    float2 bUV = (p + float2(-1, 0) + 0.5) / res;

    float r = SAMPLE_TEXTURE2D(_MainTex, sampler_PointClamp, rUV).r;
    float g = SAMPLE_TEXTURE2D(_MainTex, sampler_PointClamp, gUV).g;
    float b = SAMPLE_TEXTURE2D(_MainTex, sampler_PointClamp, bUV).b;

    return float4(r, g, b, 1);
}
扫描线
float ScanlineFactor(float2 uv)
{
    float2 res = float2(320, 180);
    float y = floor(uv.y * res.y);

    // 偶数行略暗
    float line = fmod(y, 2.0);
    return lerp(0.82, 1.0, line);
}
像素热浪
float4 PixelHeatWave(float2 uv)
{
    float2 res = float2(320, 180);
    float2 p = floor(uv * res);

    float2 noiseUV = p / 64.0 + float2(_Time.y * 0.12, _Time.y * 0.03);
    float n = SAMPLE_TEXTURE2D(_NoiseTex, sampler_PointRepeat, noiseUV).r;

    float offsetX = round((n - 0.5) * _StrengthPixels);
    float2 sampleUV = (p + float2(offsetX, 0) + 0.5) / res;

    return SAMPLE_TEXTURE2D(_MainTex, sampler_PointClamp, sampleUV);
}
